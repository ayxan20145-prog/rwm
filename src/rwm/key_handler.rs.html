16/// Check if the event modifiers match the binding modifiers.
17pub fn modifiers_match(event: KeyButMask, binding: ModMask) -> bool {
18    event.contains(KeyButMask::MOD4) == binding.contains(ModMask::M4)
19        && event.contains(KeyButMask::SHIFT) == binding.contains(ModMask::SHIFT)
20        && event.contains(KeyButMask::CONTROL) == binding.contains(ModMask::CONTROL)
21        && event.contains(KeyButMask::MOD1) == binding.contains(ModMask::M1)
22}

24/// Handle a KeyPress event by executing the matching action.
25pub fn handle_key_press<C: Connection>(
26    conn: &C,
27    event: &KeyPressEvent,
28    bindings: &[KeyBinding],
29    workspaces: &mut [Workspace],
30    current: &mut usize,
31    focused: &mut Option<Window>,
32    fullscreen_states: &mut Vec<FullscreenState>,
33    screen: &Screen,
34    bar: &bar::Bar,
35    show_bar: &mut bool,
36) -> Result<(), Box<dyn std::error::Error>> {
37    for binding in bindings {
38        if event.detail == binding.key && modifiers_match(event.state, binding.modifiers) {
39            match binding.action {
40                Action::Exit => std::process::exit(0),
41
42                Action::Close => {
43                    if let Some(window) = *focused {
44                        conn.kill_client(window)?;
45                        conn.flush()?;
46                    }
47                }
48
49                Action::Move(dir, amount) => {
50                    if let Some(window) = *focused {
51                        if is_floating(&workspaces[*current], window) {
52                            let geom = conn.get_geometry(window)?.reply()?;
53                            let (dx, dy) = match dir {
54                                Direction::Left  => (-amount, 0),
55                                Direction::Right => (amount, 0),
56                                Direction::Up    => (0, -amount),
57                                Direction::Down  => (0, amount),
58                            };
59                            move_window(conn, window, geom.x as i32 + dx, geom.y as i32 + dy)?;
60                        }
61                    }
62                }
63
64                Action::Resize(dir, amount) => {
65                    if let Some(window)  = *focused {
66                        if is_floating(&workspaces[*current], window) {
67                            let geom = conn.get_geometry(window)?.reply()?;
68                            let (new_w, new_h) = match dir {
69                                Direction::Left => (geom.width.saturating_sub(amount), geom.height),
70                                Direction::Right => (geom.width.saturating_add(amount), geom.height),
71                                Direction::Down => (geom.width, geom.height.saturating_add(amount)),
72                                Direction::Up => (geom.width, geom.height.saturating_sub(amount)),
73                            };
74                            resize_window(conn, window, new_w as u32, new_h as u32)?;
75                        }
76                    }
77                }
78
79                Action::Fullscreen => {
80                    if let Some(window) = *focused {
81                        fullscreen(conn, fullscreen_states, window, screen)?;
82                    }
83                }
84
85                Action::Workspace(ws) => switch_workspace(conn, workspaces, current, (ws - 1) as usize, focused, screen, *show_bar)?,
86
87                Action::MoveToWorkspace(ws) => {
88                    if let Some(window) = *focused {
89                        move_to_workspace(conn, workspaces, *current, (ws - 1) as usize, window, screen, *show_bar)?;
90                    }
91                }
92
93                Action::Focus(dir) => {
94                    match dir {
95                        Direction::Left  => focus_prev(conn, &workspaces[*current], focused)?,
96                        Direction::Right => focus_next(conn, &workspaces[*current], focused)?,
97                        _ => {}
98                    }
99                }
100
101                Action::ToggleBar => {
102                    *show_bar = !*show_bar;
103                    if *show_bar {
104                        conn.map_window(bar.window)?;
105                    } else {
106                        conn.unmap_window(bar.window)?;
107                    }
108                    conn.flush()?;
109                }
110
111                Action::VolumeDown => {
112                    Command::new("sh").arg("-c").arg(VOLUME_DOWN_CMD).spawn()?;
113                }
114                Action::VolumeUp => {
115                    Command::new("sh").arg("-c").arg(VOLUME_UP_CMD).spawn()?;
116                }
117                Action::Mute => {
118                    Command::new("sh").arg("-c").arg(VOLUME_MUTE_CMD).spawn()?;
119                }
120
121                Action::ToggleFloating => {
122                    if let Some(window) = *focused {
123                        toggle_floating(&mut workspaces[*current], window);
124                        layout::tile(conn, &workspaces[*current], screen, *show_bar)?;
125                    }
126                }
127                Action::Run(cmd) => {
128                    Command::new("sh").arg("-c").arg(cmd).spawn()?;
129                }
130            }
131            break;
132        }
133    }
134    conn.flush()?;
135    Ok(())
136}
